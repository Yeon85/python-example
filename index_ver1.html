Chapter 5. CUI로 미니 게임 만들기

# 숫자 맞추기 게임
import random
num = random.randint(1, 10)
guess = int(input("숫자를 맞혀보세요: "))
if guess == num:
    print("정답!")
else:
    print("틀렸습니다. 정답은", num)

Chapter 7. GUI 기초 2

# 버튼 클릭 이벤트
import tkinter as tk

def say_hello():
    label.config(text="안녕하세요!")

root = tk.Tk()
label = tk.Label(root, text="")
label.pack()
btn = tk.Button(root, text="클릭", command=say_hello)
btn.pack()
root.mainloop()
